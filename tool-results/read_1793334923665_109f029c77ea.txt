    61→    61→/** Recursively collect image URLs from an unknown MCP payload. */
    62→    62→function collectImages(value: unknown, depth = 0, seen = new Set<string>()): { url: string; label?: string }[] {
    63→    63→  if (depth > 8 || seen.size > 40) return [];
    64→    64→  if (typeof value === "string") {
    65→    65→    if (/^https:\/\/\S+\.(png|jpe?g|webp|gif)(\?\S*)?$/i.test(value) && !seen.has(value)) {
    66→    66→      seen.add(value);
    67→    67→      return [{ url: value }];
    68→    68→    }
    69→    69→    return [];
    70→    70→  }
    71→    71→  if (Array.isArray(value)) {
    72→    72→    return value.flatMap((v) => collectImages(v, depth + 1, seen));
    73→    73→  }
    74→    74→  if (value && typeof value === "object") {
    75→    75→    const out: { url: string; label?: string }[] = [];
    76→    76→    const obj = value as Record<string, unknown>;
    77→    77→    if (typeof obj.url === "string" && /^https/i.test(obj.url) && /\.(png|jpe?g|webp|gif)(\?\S*)?$/i.test(obj.url) && !seen.has(obj.url)) {
    78→    78→      seen.add(obj.url);
    79→    79→      out.push({ url: obj.url, label: typeof obj.size === "string" ? obj.size : typeof obj.language === "string" ? obj.language : undefined });
    80→    80→    }
    81→    81→    for (const v of Object.values(obj)) out.push(...collectImages(v, depth + 1, seen));
    82→    82→    return out;
    83→    83→  }
    84→    84→  return [];
    85→    85→}
    86→    86→
    87→    87→/** Collect temp-page preview URLs. */
    88→    88→function collectPageUrls(value: unknown, depth = 0, seen = new Set<string>()): string[] {
    89→    89→  if (depth > 8 || seen.size > 20) return [];
    90→    90→  if (Array.isArray(value)) return value.flatMap((v) => collectPageUrls(v, depth + 1, seen));
    91→    91→  if (value && typeof value === "object") {
    92→    92→    const out: string[] = [];
    93→    93→    const obj = value as Record<string, unknown>;
    94→    94→    if (typeof obj.pageUrl === "string" && obj.pageUrl.startsWith("http") && !seen.has(obj.pageUrl)) {
    95→    95→      seen.add(obj.pageUrl);
    96→    96→      out.push(obj.pageUrl);
    97→    97→    }
    98→    98→    for (const v of Object.values(obj)) out.push(...collectPageUrls(v, depth + 1, seen));
    99→    99→    return out;
   100→   100→  }
   101→   101→  return [];
   102→   102→}

   104→   104→function stringifyPayload(v: unknown): string {
   105→   105→  try {
   106→   106→    return JSON.stringify(v, null, 2);
   107→   107→  } catch {
   108→   108→    return String(v);
   109→   109→  }
