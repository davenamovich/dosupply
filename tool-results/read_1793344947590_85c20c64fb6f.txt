    72→    72→/** Recursively collect image URLs from an unknown MCP payload. */
    73→    73→function collectImages(value: unknown, depth = 0, seen = new Set<string>()): { url: string; label?: string }[] {
    74→    74→  if (depth > 8 || seen.size > 40) return [];
    75→    75→  if (typeof value === "string") {
    76→    76→    if (/^https:\/\/\S+\.(png|jpe?g|webp|gif)(\?\S*)?$/i.test(value) && !seen.has(value)) {
    77→    77→      seen.add(value);
    78→    78→      return [{ url: value }];
    79→    79→    }
    80→    80→    return [];
    81→    81→  }
    82→    82→  if (Array.isArray(value)) {
    83→    83→    return value.flatMap((v) => collectImages(v, depth + 1, seen));
    84→    84→  }
    85→    85→  if (value && typeof value === "object") {
    86→    86→    const out: { url: string; label?: string }[] = [];
    87→    87→    const obj = value as Record<string, unknown>;
    88→    88→    if (typeof obj.url === "string" && /^https/i.test(obj.url) && /\.(png|jpe?g|webp|gif)(\?\S*)?$/i.test(obj.url) && !seen.has(obj.url)) {
    89→    89→      seen.add(obj.url);
    90→    90→      out.push({ url: obj.url, label: typeof obj.size === "string" ? obj.size : typeof obj.language === "string" ? obj.language : undefined });
    91→    91→    }
    92→    92→    for (const v of Object.values(obj)) out.push(...collectImages(v, depth + 1, seen));
    93→    93→    return out;
    94→    94→  }
    95→    95→  return [];
    96→    96→}
    97→    97→
    98→    98→/** Collect temp-page preview URLs. */
    99→    99→function collectPageUrls(value: unknown, depth = 0, seen = new Set<string>()): string[] {
   100→   100→  if (depth > 8 || seen.size > 20) return [];
   101→   101→  if (Array.isArray(value)) return value.flatMap((v) => collectPageUrls(v, depth + 1, seen));
   102→   102→  if (value && typeof value === "object") {
   103→   103→    const out: string[] = [];
   104→   104→    const obj = value as Record<string, unknown>;
   105→   105→    if (typeof obj.pageUrl === "string" && obj.pageUrl.startsWith("http") && !seen.has(obj.pageUrl)) {
   106→   106→      seen.add(obj.pageUrl);
   107→   107→      out.push(obj.pageUrl);
   108→   108→    }
   109→   109→    for (const v of Object.values(obj)) out.push(...collectPageUrls(v, depth + 1, seen));
   110→   110→    return out;
   111→   111→  }
   112→   112→  return [];
   113→   113→}
