    65→    65→/** Recursively collect image URLs from an unknown MCP payload. */
    66→    66→function collectImages(value: unknown, depth = 0, seen = new Set<string>()): { url: string; label?: string }[] {
    67→    67→  if (depth > 8 || seen.size > 40) return [];
    68→    68→  if (typeof value === "string") {
    69→    69→    if (/^https:\/\/\S+\.(png|jpe?g|webp|gif)(\?\S*)?$/i.test(value) && !seen.has(value)) {
    70→    70→      seen.add(value);
    71→    71→      return [{ url: value }];
    72→    72→    }
    73→    73→    return [];
    74→    74→  }
    75→    75→  if (Array.isArray(value)) {
    76→    76→    return value.flatMap((v) => collectImages(v, depth + 1, seen));
    77→    77→  }
    78→    78→  if (value && typeof value === "object") {
    79→    79→    const out: { url: string; label?: string }[] = [];
    80→    80→    const obj = value as Record<string, unknown>;
    81→    81→    if (typeof obj.url === "string" && /^https/i.test(obj.url) && /\.(png|jpe?g|webp|gif)(\?\S*)?$/i.test(obj.url) && !seen.has(obj.url)) {
    82→    82→      seen.add(obj.url);
    83→    83→      out.push({ url: obj.url, label: typeof obj.size === "string" ? obj.size : typeof obj.language === "string" ? obj.language : undefined });
    84→    84→    }
    85→    85→    for (const v of Object.values(obj)) out.push(...collectImages(v, depth + 1, seen));
    86→    86→    return out;
    87→    87→  }
    88→    88→  return [];
    89→    89→}
    90→    90→
    91→    91→/** Collect temp-page preview URLs. */
    92→    92→function collectPageUrls(value: unknown, depth = 0, seen = new Set<string>()): string[] {
    93→    93→  if (depth > 8 || seen.size > 20) return [];
    94→    94→  if (Array.isArray(value)) return value.flatMap((v) => collectPageUrls(v, depth + 1, seen));
    95→    95→  if (value && typeof value === "object") {
    96→    96→    const out: string[] = [];
    97→    97→    const obj = value as Record<string, unknown>;
    98→    98→    if (typeof obj.pageUrl === "string" && obj.pageUrl.startsWith("http") && !seen.has(obj.pageUrl)) {
    99→    99→      seen.add(obj.pageUrl);
   100→   100→      out.push(obj.pageUrl);
   101→   101→    }
   102→   102→    for (const v of Object.values(obj)) out.push(...collectPageUrls(v, depth + 1, seen));
   103→   103→    return out;
   104→   104→  }
   105→   105→  return [];
   106→   106→}
