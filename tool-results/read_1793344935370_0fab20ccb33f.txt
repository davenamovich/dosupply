    72→/** Recursively collect image URLs from an unknown MCP payload. */
    73→function collectImages(value: unknown, depth = 0, seen = new Set<string>()): { url: string; label?: string }[] {
    74→  if (depth > 8 || seen.size > 40) return [];
    75→  if (typeof value === "string") {
    76→    if (/^https:\/\/\S+\.(png|jpe?g|webp|gif)(\?\S*)?$/i.test(value) && !seen.has(value)) {
    77→      seen.add(value);
    78→      return [{ url: value }];
    79→    }
    80→    return [];
    81→  }
    82→  if (Array.isArray(value)) {
    83→    return value.flatMap((v) => collectImages(v, depth + 1, seen));
    84→  }
    85→  if (value && typeof value === "object") {
    86→    const out: { url: string; label?: string }[] = [];
    87→    const obj = value as Record<string, unknown>;
    88→    if (typeof obj.url === "string" && /^https/i.test(obj.url) && /\.(png|jpe?g|webp|gif)(\?\S*)?$/i.test(obj.url) && !seen.has(obj.url)) {
    89→      seen.add(obj.url);
    90→      out.push({ url: obj.url, label: typeof obj.size === "string" ? obj.size : typeof obj.language === "string" ? obj.language : undefined });
    91→    }
    92→    for (const v of Object.values(obj)) out.push(...collectImages(v, depth + 1, seen));
    93→    return out;
    94→  }
    95→  return [];
    96→}
    97→
    98→/** Collect temp-page preview URLs. */
    99→function collectPageUrls(value: unknown, depth = 0, seen = new Set<string>()): string[] {
   100→  if (depth > 8 || seen.size > 20) return [];
   101→  if (Array.isArray(value)) return value.flatMap((v) => collectPageUrls(v, depth + 1, seen));
   102→  if (value && typeof value === "object") {
   103→    const out: string[] = [];
   104→    const obj = value as Record<string, unknown>;
   105→    if (typeof obj.pageUrl === "string" && obj.pageUrl.startsWith("http") && !seen.has(obj.pageUrl)) {
   106→      seen.add(obj.pageUrl);
   107→      out.push(obj.pageUrl);
   108→    }
   109→    for (const v of Object.values(obj)) out.push(...collectPageUrls(v, depth + 1, seen));
   110→    return out;
   111→  }
   112→  return [];
   113→}

   118→  } catch {
   119→    return String(v);

   127→function useKeys() {
   128→  const [keys, setKeys] = useState<KeyStatus[] | null>(null);

   134→  }, []);
   135→  useEffect(() => {
   136→    refresh();
   137→  }, [refresh]);
   138→  return { keys, refresh };

   793→  const [mp4, setMp4] = useState<Mp4State | null>(null);
   794→  const [mp4Busy, setMp4Busy] = useState(false);

   796→  const mp4PollRef = useRef<ReturnType<typeof setInterval> | null>(null);
   797→
   798→  const stopPolling = useCallback(() => {
   799→    if (pollRef.current) {
   800→      clearInterval(pollRef.current);
   801→      pollRef.current = null;

   806→    if (mp4PollRef.current) {
   807→      clearInterval(mp4PollRef.current);
   808→      mp4PollRef.current = null;
   809→    }
   810→  }, []);
   811→
   812→  useEffect(
   813→    () => () => {
   814→      stopPolling();
   815→      stopMp4Polling();

   820→  function startPolling(jobId: string) {

   825→        pageUrl?: string | null;

   830→        prev
   831→          ? {
   832→              ...prev,
   833→              status: data.status ?? prev.status,
   834→              pageUrl: data.pageUrl ?? prev.pageUrl,
   835→              error: data.error ?? prev.error,
   836→            }
   837→          : prev
   838→      );
   839→      if (data.status === "done" || data.status === "failed") stopPolling();
   840→    }, 15_000);
   841→  }
   842→
   843→  function startMp4Polling(jobId: string) {
   844→    stopMp4Polling();

   846→      const { ok, data } = await postJson<{
   847→        status?: string;
   848→        mp4Url?: string | null;
   849→        error?: string | null;
   850→      }>("/api/mcp/job", { kind: "mp4", jobId });

   852→      setMp4((prev) =>
   853→        prev
   854→          ? {
   855→              ...prev,
   856→              status: data.status ?? prev.status,
   857→              mp4Url: data.mp4Url ?? prev.mp4Url,
   858→              error: data.error ?? prev.error,
   859→            }
   860→          : prev
   861→      );
   862→      if (data.status === "done" || data.status === "failed") stopMp4Polling();
   863→    }, 15_000);
