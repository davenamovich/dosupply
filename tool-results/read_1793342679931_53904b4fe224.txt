    65→/** Recursively collect image URLs from an unknown MCP payload. */
    66→function collectImages(value: unknown, depth = 0, seen = new Set<string>()): { url: string; label?: string }[] {
    67→  if (depth > 8 || seen.size > 40) return [];
    68→  if (typeof value === "string") {
    69→    if (/^https:\/\/\S+\.(png|jpe?g|webp|gif)(\?\S*)?$/i.test(value) && !seen.has(value)) {
    70→      seen.add(value);
    71→      return [{ url: value }];
    72→    }
    73→    return [];
    74→  }
    75→  if (Array.isArray(value)) {
    76→    return value.flatMap((v) => collectImages(v, depth + 1, seen));
    77→  }
    78→  if (value && typeof value === "object") {
    79→    const out: { url: string; label?: string }[] = [];
    80→    const obj = value as Record<string, unknown>;
    81→    if (typeof obj.url === "string" && /^https/i.test(obj.url) && /\.(png|jpe?g|webp|gif)(\?\S*)?$/i.test(obj.url) && !seen.has(obj.url)) {
    82→      seen.add(obj.url);
    83→      out.push({ url: obj.url, label: typeof obj.size === "string" ? obj.size : typeof obj.language === "string" ? obj.language : undefined });
    84→    }
    85→    for (const v of Object.values(obj)) out.push(...collectImages(v, depth + 1, seen));
    86→    return out;
    87→  }
    88→  return [];
    89→}
    90→
    91→/** Collect temp-page preview URLs. */
    92→function collectPageUrls(value: unknown, depth = 0, seen = new Set<string>()): string[] {
    93→  if (depth > 8 || seen.size > 20) return [];
    94→  if (Array.isArray(value)) return value.flatMap((v) => collectPageUrls(v, depth + 1, seen));
    95→  if (value && typeof value === "object") {
    96→    const out: string[] = [];
    97→    const obj = value as Record<string, unknown>;
    98→    if (typeof obj.pageUrl === "string" && obj.pageUrl.startsWith("http") && !seen.has(obj.pageUrl)) {
    99→      seen.add(obj.pageUrl);
   100→      out.push(obj.pageUrl);
   101→    }
   102→    for (const v of Object.values(obj)) out.push(...collectPageUrls(v, depth + 1, seen));
   103→    return out;
   104→  }
   105→  return [];
   106→}

   108→function stringifyPayload(v: unknown): string {

   110→    return JSON.stringify(v, null, 2);
   111→  } catch {
   112→    return String(v);
   113→  }

   120→function useKeys() {
   121→  const [keys, setKeys] = useState<KeyStatus[] | null>(null);
   122→  const refresh = useCallback(() => {
   123→    fetch("/api/keys")
   124→      .then((r) => r.json())
   125→      .then((d) => setKeys(d.keys ?? []))
   126→      .catch(() => {});
   127→  }, []);
   128→  useEffect(() => {
   129→    refresh();
   130→  }, [refresh]);
   131→  return { keys, refresh };
   132→}

   361→  async function generate(inputArg?: string) {

   368→      input: topic,
   369→      kind,
   370→      format,

   783→  const stopPolling = useCallback(() => {
   784→    if (pollRef.current) {

   797→  useEffect(
   798→    () => () => {

   820→              error: data.error ?? prev.error,
   821→            }

   832→        status?: string;
   833→        mp4Url?: string | null;
