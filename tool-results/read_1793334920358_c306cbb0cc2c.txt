    61→/** Recursively collect image URLs from an unknown MCP payload. */
    62→function collectImages(value: unknown, depth = 0, seen = new Set<string>()): { url: string; label?: string }[] {
    63→  if (depth > 8 || seen.size > 40) return [];
    64→  if (typeof value === "string") {
    65→    if (/^https:\/\/\S+\.(png|jpe?g|webp|gif)(\?\S*)?$/i.test(value) && !seen.has(value)) {
    66→      seen.add(value);
    67→      return [{ url: value }];
    68→    }
    69→    return [];
    70→  }
    71→  if (Array.isArray(value)) {
    72→    return value.flatMap((v) => collectImages(v, depth + 1, seen));
    73→  }
    74→  if (value && typeof value === "object") {
    75→    const out: { url: string; label?: string }[] = [];
    76→    const obj = value as Record<string, unknown>;
    77→    if (typeof obj.url === "string" && /^https/i.test(obj.url) && /\.(png|jpe?g|webp|gif)(\?\S*)?$/i.test(obj.url) && !seen.has(obj.url)) {
    78→      seen.add(obj.url);
    79→      out.push({ url: obj.url, label: typeof obj.size === "string" ? obj.size : typeof obj.language === "string" ? obj.language : undefined });
    80→    }
    81→    for (const v of Object.values(obj)) out.push(...collectImages(v, depth + 1, seen));
    82→    return out;
    83→  }
    84→  return [];
    85→}
    86→
    87→/** Collect temp-page preview URLs. */
    88→function collectPageUrls(value: unknown, depth = 0, seen = new Set<string>()): string[] {
    89→  if (depth > 8 || seen.size > 20) return [];
    90→  if (Array.isArray(value)) return value.flatMap((v) => collectPageUrls(v, depth + 1, seen));
    91→  if (value && typeof value === "object") {
    92→    const out: string[] = [];
    93→    const obj = value as Record<string, unknown>;
    94→    if (typeof obj.pageUrl === "string" && obj.pageUrl.startsWith("http") && !seen.has(obj.pageUrl)) {
    95→      seen.add(obj.pageUrl);
    96→      out.push(obj.pageUrl);
    97→    }
    98→    for (const v of Object.values(obj)) out.push(...collectPageUrls(v, depth + 1, seen));
    99→    return out;
   100→  }
   101→  return [];
   102→}

   104→function stringifyPayload(v: unknown): string {
   105→  try {
   106→    return JSON.stringify(v, null, 2);
   107→  } catch {
   108→    return String(v);
   109→  }
   110→}

   125→    refresh();
   126→  }, [refresh]);
   127→  return { keys, refresh };

   620→function AudioTab({ mcpLive }: { mcpLive: boolean }) {
   621→  const [topic, setTopic] = useState("Why downtime waits are the real cost of legacy automation parts");
   622→  const [starting, setStarting] = useState(false);
   623→  const [audio, setAudio] = useState<AudioState | null>(null);
   624→  const pollRef = useRef<ReturnType<typeof setInterval> | null>(null);
   625→
   626→  const stopPolling = useCallback(() => {
   627→    if (pollRef.current) {
   628→      clearInterval(pollRef.current);
   629→      pollRef.current = null;
   630→    }
   631→  }, []);
   632→
   633→  useEffect(() => stopPolling, [stopPolling]);
   634→
   635→  async function start() {
   636→    setStarting(true);
   637→    setAudio({ jobId: null, status: "starting", pageUrl: null, mp4Url: null, error: null });
   638→    const { ok, data } = await postJson<{
   639→      mode?: string;
   640→      jobId?: string | null;
   641→      status?: string;
   642→      pageUrl?: string | null;
   643→      error?: string;
   644→      code?: string;
   645→    }>("/api/mcp/audio", { topic });
   646→    if (ok && data.mode === "live") {
   647→      const st: AudioState = {
   648→        jobId: data.jobId ?? null,
   649→        status: data.status ?? "queued",
   650→        pageUrl: data.pageUrl ?? null,
   651→        mp4Url: null,
   652→        error: null,
   653→      };
   654→      setAudio(st);
   655→      if (st.jobId) startPolling(st.jobId);
   656→    } else {
   657→      setAudio({
   658→        jobId: null,
   659→        status: null,
   660→        pageUrl: null,
   661→        mp4Url: null,
   662→        error: (data.error as string) || "create_one_minute failed.",
   663→      });
   664→    }
   665→    setStarting(false);
   666→  }
   667→
   668→  function startPolling(jobId: string) {
   669→    stopPolling();
   670→    pollRef.current = setInterval(async () => {
   671→      const { ok, data } = await postJson<{
   672→        status?: string;
   673→        pageUrl?: string | null;
   674→        mp4Url?: string | null;
   675→        error?: string | null;
   676→      }>("/api/mcp/job", { kind: "one_minute", jobId });
   677→      if (!ok) return; // transient — keep polling
   678→      setAudio((prev) =>
   679→        prev
   680→          ? {
   681→              ...prev,
   682→              status: data.status ?? prev.status,
   683→              pageUrl: data.pageUrl ?? prev.pageUrl,
   684→              mp4Url: data.mp4Url ?? prev.mp4Url,
   685→              error: data.error ?? prev.error,
   686→            }
   687→          : prev
   688→      );
   689→      if (data.status === "done" || data.status === "failed") stopPolling();
   690→    }, 15_000);
   691→  }
   692→
   693→  const done = audio?.status === "done";
   694→  const failed = audio?.status === "failed";
   695→
   696→  return (
   697→    <div className="panel rounded-2xl p-5 sm:p-6">
   698→      <div className="flex flex-col gap-3 sm:flex-row">
   699→        <Input
   700→          value={topic}
   701→          onChange={(e) => setTopic(e.target.value)}
   702→          placeholder="Topic for the ~60s voiced take…"
   703→          className="h-11 flex-1 border-white/15 bg-black/50 font-mono text-sm text-zinc-100 placeholder:text-zinc-600 focus-visible:ring-amber-400/50"
   704→          aria-label="Audio take topic"
   705→        />
   706→        <Button
   707→          onClick={start}
   708→          disabled={starting || !topic.trim() || (audio?.jobId != null && !done && !failed)}
   709→          className="h-11 bg-teal-400 font-bold text-black hover:bg-teal-300"
   710→        >
   711→          {starting ? <RefreshCw className="h-4 w-4 animate-spin" /> : <AudioLines className="h-4 w-4" />}
   712→          <span className="ml-2">Start ~60s Take</span>
   713→        </Button>
   714→      </div>
   715→
   716→      <p className="mt-3 font-mono text-[10px] text-zinc-600">
   717→        tool: create_one_minute → get_job_status polled every 15s · the engine returns a
   718→        captioned page preview; an MP4 is a separate approval-gated step we never
   719→        auto-trigger{!mcpLive && " · BYOK key not set yet (see BYOK Setup)"}
   720→      </p>
   721→
   722→      {audio && (
   723→        <div
   724→          className={cn(
   725→            "mt-5 rounded-xl border p-4",
   726→            failed ? "border-red-400/30 bg-red-400/5" : done ? "border-teal-400/25 bg-teal-400/5" : "border-white/10 bg-black/30"
   727→          )}
   728→        >
   729→          {audio.error ? (
   730→            <p className="text-sm text-red-300">{audio.error}</p>
   731→          ) : (
   732→            <>
   733→              <div className="flex flex-wrap items-center gap-2">
   734→                <span
   735→                  className={cn(
   736→                    "h-2 w-2 rounded-full",
   737→                    done ? "bg-teal-400" : failed ? "bg-red-400" : "bg-amber-400 amber-dot"
   738→                  )}
   739→                />
   740→                <span className="font-mono text-xs uppercase tracking-widest text-zinc-300">
   741→                  job status: {audio.status ?? "unknown"}
   742→                </span>
   743→                {audio.jobId && (
   744→                  <span className="font-mono text-[10px] text-zinc-600">jobId: {audio.jobId}</span>
   745→                )}
   746→              </div>
   747→              {(audio.status === "rendering" || audio.status === "queued" || audio.status === "starting") && (
   748→                <p className="mt-2 text-xs text-amber-200/80">
   749→                  The engine is writing and voicing the take — polling every 15s. This can take
   750→                  a few minutes; the page keeps polling until done or failed.

   753→              {audio.pageUrl && (
   754→                <a
   755→                  href={audio.pageUrl}
   756→                  target="_blank"
   757→                  rel="noopener noreferrer"
   758→                  className="mt-3 inline-flex items-center gap-1.5 rounded-lg border border-amber-400/40 bg-amber-400/10 px-3.5 py-2 text-sm font-medium text-amber-200 hover:bg-amber-400/20"
   759→                >
   760→                  <ExternalLink className="h-3.5 w-3.5" />
   761→                  {done ? "Open the finished captioned page" : "Open the in-progress preview page"}
   762→                </a>
   763→              )}
   764→              {audio.mp4Url && (
   765→                <a
   766→                  href={audio.mp4Url}
   767→                  target="_blank"
   768→                  rel="noopener noreferrer"
   769→                  className="mt-2 inline-flex items-center gap-1.5 rounded-lg border border-teal-400/40 bg-teal-400/10 px-3.5 py-2 text-sm font-medium text-teal-200 hover:bg-teal-400/20"
   770→                >
   771→                  <ExternalLink className="h-3.5 w-3.5" /> Rendered MP4 (as returned by the engine)
   772→                </a>
   773→              )}
   774→            </>
   775→          )}
   776→        </div>
   777→      )}
   778→    </div>
   779→  );

   808→      saved?: Record<string, { saved: boolean; probe?: string; account?: string }>;

   813→    });

   815→      const saved = data.saved?.[which];
   816→      const account = saved?.account ? ` Account: ${saved.account}` : "";
   817→      setMsg({
   818→        kind: "ok",
   819→        text:
   820→          which === "plunder_riffle"
   821→            ? `P&R key saved and verified against bot.plunderandriffle.com.${account} Live multimedia calls are now enabled.`
   822→            : `Exa key saved (${saved?.probe ?? "saved"}). Research now runs through api.exa.ai.`,

   847→        return;
   848→      }
   849→      setMsg({ kind: "ok", text: "Key cleared from the runtime store." });

   851→    } catch {
   852→      setMsg({ kind: "err", text: "Network error while clearing the key." });

   855→  }

   872→        <KeyCard status={prStatus} icon={<Terminal className="h-4 w-4 text-teal-300" />} />
   873→        <KeyCard status={exaStatus} icon={<Languages className="h-4 w-4 text-teal-300" />} />
   874→      </div>
   875→
   876→      <div className="mt-5 space-y-4">
   877→        <div>
   878→          <label className="kicker mb-1.5 block" htmlFor="setup-passcode">
   879→            Setup passcode
   880→          </label>
   881→          <Input
   882→            id="setup-passcode"
   883→            type="password"
   884→            value={passcode}
   885→            onChange={(e) => setPasscode(e.target.value)}
